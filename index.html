CREATE EXTENSION IF NOT EXISTS pgcrypto; -- Core organizational tables CREATE TABLE IF NOT EXISTS public.roles ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), name TEXT NOT NULL UNIQUE, description TEXT, is_system_default BOOLEAN DEFAULT false, created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now() ); CREATE TABLE IF NOT EXISTS public.departments ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), name TEXT NOT NULL, code TEXT, location TEXT, description TEXT, status TEXT DEFAULT 'active' CHECK (status IN ('active', 'inactive')), created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now() ); CREATE TABLE IF NOT EXISTS public.profiles ( id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE, full_name TEXT, email TEXT, employee_code TEXT, department_id UUID REFERENCES public.departments(id), role TEXT NOT NULL DEFAULT 'Auditor / Read Only', job_title TEXT, phone TEXT, status TEXT DEFAULT 'active', avatar_url TEXT, created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now() ); -- Document control foundation CREATE TABLE IF NOT EXISTS public.documents ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), document_number TEXT, title TEXT NOT NULL, department_id UUID REFERENCES public.departments(id), category TEXT NOT NULL CHECK (category IN ('Policy', 'Manual', 'SOP', 'Work Instruction', 'Form', 'Specification', 'External Document', 'Record')), revision INTEGER DEFAULT 0, issue_date DATE, effective_date DATE, review_date DATE, owner TEXT, approver TEXT, status TEXT DEFAULT 'Draft' CHECK (status IN ('Draft', 'Under Review', 'Approved', 'Obsolete')), file_url TEXT, description TEXT, created_by UUID REFERENCES auth.users(id), updated_by UUID REFERENCES auth.users(id), created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now() ); CREATE TABLE IF NOT EXISTS public.document_versions ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), document_id UUID NOT NULL REFERENCES public.documents(id) ON DELETE CASCADE, version_number INTEGER NOT NULL, file_url TEXT, comments TEXT, created_by UUID REFERENCES auth.users(id), created_at TIMESTAMPTZ DEFAULT now() ); CREATE TABLE IF NOT EXISTS public.document_approvals ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), document_id UUID NOT NULL REFERENCES public.documents(id) ON DELETE CASCADE, reviewer UUID REFERENCES auth.users(id), approver UUID REFERENCES auth.users(id), decision TEXT CHECK (decision IN ('Approved', 'Rejected', 'Pending')) DEFAULT 'Pending', comments TEXT, approved_at TIMESTAMPTZ, created_at TIMESTAMPTZ DEFAULT now() ); CREATE TABLE IF NOT EXISTS public.records ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), record_number TEXT, record_type TEXT, title TEXT, department_id UUID REFERENCES public.departments(id), status TEXT DEFAULT 'active', created_by UUID REFERENCES auth.users(id), updated_by UUID REFERENCES auth.users(id), created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now() ); -- FSMS foundation tables CREATE TABLE IF NOT EXISTS public.ncrs ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), ncr_number TEXT, date DATE, department_id UUID REFERENCES public.departments(id), source TEXT, description TEXT, severity TEXT, immediate_correction TEXT, responsible_person TEXT, status TEXT DEFAULT 'Open' CHECK (status IN ('Open', 'In Progress', 'Pending Verification', 'Closed', 'Overdue')), attachment_url TEXT, created_by UUID REFERENCES auth.users(id), created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now() ); CREATE TABLE IF NOT EXISTS public.capas ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), capa_number TEXT, ncr_id UUID REFERENCES public.ncrs(id), root_cause TEXT, corrective_action TEXT, preventive_action TEXT, responsible_person TEXT, target_date DATE, completion_date DATE, effectiveness_verification TEXT, status TEXT DEFAULT 'Open' CHECK (status IN ('Open', 'In Progress', 'Pending Verification', 'Closed', 'Overdue')), created_by UUID REFERENCES auth.users(id), created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now() ); CREATE TABLE IF NOT EXISTS public.suppliers ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), name TEXT NOT NULL, supplier_type TEXT, material_or_service TEXT, risk_level TEXT, approval_status TEXT, evaluation_score NUMERIC(5,2), evaluation_date DATE, next_evaluation_date DATE, documents_url TEXT, certificates_url TEXT, remarks TEXT, created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now() ); CREATE TABLE IF NOT EXISTS public.calibrations ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), equipment_name TEXT, equipment_id TEXT, location TEXT, instrument_type TEXT, manufacturer TEXT, model TEXT, serial_number TEXT, calibration_date DATE, next_due_date DATE, calibration_company TEXT, certificate_number TEXT, certificate_file_url TEXT, status TEXT DEFAULT 'Valid' CHECK (status IN ('Valid', 'Due Soon', 'Overdue')), remarks TEXT, created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now() ); CREATE TABLE IF NOT EXISTS public.trainings ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), employee_name TEXT, department_id UUID REFERENCES public.departments(id), training_topic TEXT, trainer TEXT, training_date DATE, assessment_score NUMERIC(5,2), result TEXT, certificate_url TEXT, expiry_date DATE, remarks TEXT, created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now() ); CREATE TABLE IF NOT EXISTS public.audit_findings ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), audit_number TEXT, finding_title TEXT, department_id UUID REFERENCES public.departments(id), severity TEXT, status TEXT DEFAULT 'Open' CHECK (status IN ('Open', 'In Progress', 'Closed', 'Overdue')), due_date DATE, created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now() ); CREATE TABLE IF NOT EXISTS public.notifications ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), user_id UUID REFERENCES auth.users(id), title TEXT NOT NULL, message TEXT, type TEXT, status TEXT DEFAULT 'unread' CHECK (status IN ('unread', 'read', 'archived')), created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now() ); CREATE TABLE IF NOT EXISTS public.activity_logs ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), user_id UUID REFERENCES auth.users(id), action TEXT NOT NULL, module TEXT, record_id UUID, timestamp TIMESTAMPTZ DEFAULT now(), details JSONB, created_at TIMESTAMPTZ DEFAULT now() ); CREATE TABLE IF NOT EXISTS public.water_ro_readings ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), reading_date DATE, reading_time TIME, parameter TEXT, result NUMERIC, unit TEXT, limit_value NUMERIC, status TEXT DEFAULT 'PASS' CHECK (status IN ('PASS', 'FAIL')), operator_name TEXT, comments TEXT, created_by UUID REFERENCES auth.users(id), created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now() );
